📥 سندات التحويل (كويري OWTR/WTR1 على مستودع 011)

الصق نتيجة الكويري كاملة (مع صف العناوين) — أعمدة: رقم السند، تاريخ التسليم، من مستودع، الى مستودع، رمزنا، اسمنا، الكمية، الوحدة، اسم الفرع، رقم فاتورة الصرف، ريمارك 2. إعادة لصق سند برقم موجود بيستبدل سطوره القديمة.
عرض/نسخ كويري SAP 
SELECT T0."DocNum" AS "رقم السند", T0."DocDate" AS "تاريخ التسليم",
T1."FromWhsCod" AS "من مستودع", T1."WhsCode" AS "الى مستودع",
T1."ItemCode" AS "رمزنا", T1."Dscription" AS "اسمنا",
T1."Quantity" AS "الكمية", T1."UomCode" AS "الوحدة",
T0."U_Branch" AS "اسم الفرع",
T0."U_ST_Remarks" AS "رقم فاتورة الصرف",
T0."Comments" AS "ريمارك 2"
FROM OWTR T0 INNER JOIN WTR1 T1 ON T0."DocEntry" = T1."DocEntry"
WHERE (T1."FromWhsCod" = '011' OR T1."WhsCode" = '011')
AND T0."DocDate" BETWEEN [%0] AND [%1]
ORDER BY T0."DocNum"

🧾 فواتير المبيعات والكلفة (كويري KLN-SAL-000004 لعملاء C0006588 وC0006751)

الصق نتيجة الكويري كاملة (مع صف العناوين). إعادة لصق فاتورة/مرتجع برقم موجود بينفس النوع بيستبدل سطوره القديمة.
عرض/نسخ كويري SAP 
/* SELECT T0."DocDate" FROM OINV T0 WHERE T0."DocDate" BETWEEN [%0] AND [%1] */
/*
  KLN-SAL-000004 (مخصصة لعملاء المؤسسة الاستهلاكية العسكرية C0006588 و C0006751)
  ======================================================================
  ربحية الفواتير والمرتجعات مع الباتش الفعلي وكلفته الحالية — نفس منطق النسخة
  الأصلية، مع فلترة إضافية على رمز العميل فقط.
  ======================================================================
*/
SELECT
    "نوع المستند", "رقم المستند", "التاريخ", "رمز العميل", "اسم العميل",
    "كود الصنف", "اسم الصنف", "الكمية",
    "سعر البيع", "خصم السطر %", "خصم الوثيقة %", "صافي السعر للوحدة",
    "صافي المبيعات",
    "رقم الديليفري المرتبط",
    "رقم الباتش",
    "كلفة الوحدة الحالية للباتش",
    CASE WHEN "رقم الباتش" IS NULL THEN "_est_cost_raw" ELSE NULL END
                                      AS "كلفة تقديرية (غير مدلفرة)",

    CASE WHEN "نوع المستند" = 'مرتجع' THEN -1 ELSE 1 END
        * "_qty_raw"
        * "كلفة الوحدة الحالية للباتش"
                                      AS "إجمالي الكلفة",

    ("صافي المبيعات" -
     (CASE WHEN "نوع المستند" = 'مرتجع' THEN -1 ELSE 1 END
        * "_qty_raw"
        * "كلفة الوحدة الحالية للباتش")
    )                                 AS "مجمل الربح",

    CASE WHEN "رقم الباتش" IS NULL THEN 'غير مدلفرة' ELSE 'مدلفرة' END
                                      AS "حالة التسليم",

    CASE
        WHEN "صافي السعر للوحدة" IS NULL OR "صافي السعر للوحدة" = 0 THEN NULL
        WHEN "كلفة الوحدة الحالية للباتش" IS NULL THEN NULL
        ELSE ROUND((("صافي السعر للوحدة" - "كلفة الوحدة الحالية للباتش") / "صافي السعر للوحدة") * 100, 2)
    END                               AS "نسبة الربح %"

FROM (

    -- ============ الفواتير (٣ مسارات للباتش) ============
    SELECT
        'فاتورة'                         AS "نوع المستند",
        T0."DocNum"                      AS "رقم المستند",
        T0."DocDate"                     AS "التاريخ",
        T0."CardCode"                    AS "رمز العميل",
        T0."CardName"                    AS "اسم العميل",
        T1."ItemCode"                    AS "كود الصنف",
        T1."Dscription"                  AS "اسم الصنف",
        T1."Quantity"                    AS "الكمية",
        T1."PriceBefDi"                  AS "سعر البيع",
        T1."DiscPrcnt"                   AS "خصم السطر %",
        T0."DiscPrcnt"                   AS "خصم الوثيقة %",
        (T1."LineTotal" * (1 - IFNULL(T0."DiscPrcnt",0)/100) / NULLIF(T1."Quantity",0))
                                          AS "صافي السعر للوحدة",

        (T1."LineTotal" * (1 - IFNULL(T0."DiscPrcnt",0)/100))
                                          AS "صافي المبيعات",

        COALESCE(
            (SELECT MAX(D."DocNum") FROM DLN1 DL
             INNER JOIN ODLN D ON DL."DocEntry" = D."DocEntry"
             WHERE DL."BaseEntry" = T0."DocEntry" AND DL."BaseLine" = T1."LineNum" AND DL."BaseType" = 13),
            (SELECT D."DocNum" FROM ODLN D
             WHERE D."DocEntry" = T1."BaseEntry" AND T1."BaseType" = 15)
        )                                 AS "رقم الديليفري المرتبط",

        COALESCE(
            (SELECT MAX(B."DistNumber")
             FROM OITL O INNER JOIN ITL1 L ON O."LogEntry" = L."LogEntry"
             INNER JOIN OBTN B ON L."ItemCode" = B."ItemCode" AND L."SysNumber" = B."SysNumber"
             WHERE O."DocType" = 13 AND O."DocEntry" = T0."DocEntry" AND O."DocLine" = T1."LineNum"
               AND O."StockEff" = 1),
            (SELECT MAX(B."DistNumber")
             FROM DLN1 DL
             INNER JOIN OITL O ON O."DocType" = 15 AND O."DocEntry" = DL."DocEntry" AND O."DocLine" = DL."LineNum"
             INNER JOIN ITL1 L ON O."LogEntry" = L."LogEntry"
             INNER JOIN OBTN B ON L."ItemCode" = B."ItemCode" AND L."SysNumber" = B."SysNumber"
             WHERE DL."BaseEntry" = T0."DocEntry" AND DL."BaseLine" = T1."LineNum" AND DL."BaseType" = 13
               AND O."StockEff" = 1),
            (SELECT MAX(B."DistNumber")
             FROM OITL O INNER JOIN ITL1 L ON O."LogEntry" = L."LogEntry"
             INNER JOIN OBTN B ON L."ItemCode" = B."ItemCode" AND L."SysNumber" = B."SysNumber"
             WHERE O."DocType" = 15 AND O."DocEntry" = T1."BaseEntry" AND O."DocLine" = T1."BaseLine"
               AND T1."BaseType" = 15 AND O."StockEff" = 1)
        )                                 AS "رقم الباتش",

        (COALESCE(
            (SELECT MAX(B."CostTotal" / NULLIF(B."Quantity",0))
             FROM OITL O INNER JOIN ITL1 L ON O."LogEntry" = L."LogEntry"
             INNER JOIN OBTN B ON L."ItemCode" = B."ItemCode" AND L."SysNumber" = B."SysNumber"
             WHERE O."DocType" = 13 AND O."DocEntry" = T0."DocEntry" AND O."DocLine" = T1."LineNum"
               AND O."StockEff" = 1),
            (SELECT MAX(B."CostTotal" / NULLIF(B."Quantity",0))
             FROM DLN1 DL
             INNER JOIN OITL O ON O."DocType" = 15 AND O."DocEntry" = DL."DocEntry" AND O."DocLine" = DL."LineNum"
             INNER JOIN ITL1 L ON O."LogEntry" = L."LogEntry"
             INNER JOIN OBTN B ON L."ItemCode" = B."ItemCode" AND L."SysNumber" = B."SysNumber"
             WHERE DL."BaseEntry" = T0."DocEntry" AND DL."BaseLine" = T1."LineNum" AND DL."BaseType" = 13
               AND O."StockEff" = 1),
            (SELECT MAX(B."CostTotal" / NULLIF(B."Quantity",0))
             FROM OITL O INNER JOIN ITL1 L ON O."LogEntry" = L."LogEntry"
             INNER JOIN OBTN B ON L."ItemCode" = B."ItemCode" AND L."SysNumber" = B."SysNumber"
             WHERE O."DocType" = 15 AND O."DocEntry" = T1."BaseEntry" AND O."DocLine" = T1."BaseLine"
               AND T1."BaseType" = 15 AND O."StockEff" = 1)
        ) * IFNULL(NULLIF(T1."NumPerMsr",0), 1))
                                          AS "كلفة الوحدة الحالية للباتش",

        ((SELECT SUM(Q."Quantity" * (BN."CostTotal" / NULLIF(BN."Quantity",0)))
                / NULLIF(SUM(Q."Quantity"),0)
         FROM OBTQ Q
         INNER JOIN OBTN BN ON Q."ItemCode" = BN."ItemCode" AND Q."SysNumber" = BN."SysNumber"
         WHERE Q."ItemCode" = T1."ItemCode" AND Q."WhsCode" = T1."WhsCode" AND Q."Quantity" > 0
        ) * IFNULL(NULLIF(T1."NumPerMsr",0), 1))
                                          AS "_est_cost_raw",

        T1."Quantity"                     AS "_qty_raw"

    FROM OINV T0
    INNER JOIN INV1 T1 ON T0."DocEntry" = T1."DocEntry"
    WHERE T0."DocDate" BETWEEN [%0] AND [%1]
      AND T0."CANCELED" = 'N'
      AND T0."CardCode" IN ('C0006588','C0006751')

    UNION ALL

    -- ============ المرتجعات (مباشرة دائماً) ============
    SELECT
        'مرتجع',
        T0."DocNum",
        T0."DocDate",
        T0."CardCode",
        T0."CardName",
        T1."ItemCode",
        T1."Dscription",
        -T1."Quantity",
        T1."PriceBefDi",
        T1."DiscPrcnt",
        T0."DiscPrcnt",
        -(T1."LineTotal" * (1 - IFNULL(T0."DiscPrcnt",0)/100) / NULLIF(T1."Quantity",0)),

        -(T1."LineTotal" * (1 - IFNULL(T0."DiscPrcnt",0)/100)),

        NULL,

        (SELECT MAX(B."DistNumber")
         FROM OITL O INNER JOIN ITL1 L ON O."LogEntry" = L."LogEntry"
         INNER JOIN OBTN B ON L."ItemCode" = B."ItemCode" AND L."SysNumber" = B."SysNumber"
         WHERE O."DocType" = 14 AND O."DocEntry" = T0."DocEntry" AND O."DocLine" = T1."LineNum"
           AND O."StockEff" = 1),

        (SELECT MAX(B."CostTotal" / NULLIF(B."Quantity",0))
         FROM OITL O INNER JOIN ITL1 L ON O."LogEntry" = L."LogEntry"
         INNER JOIN OBTN B ON L."ItemCode" = B."ItemCode" AND L."SysNumber" = B."SysNumber"
         WHERE O."DocType" = 14 AND O."DocEntry" = T0."DocEntry" AND O."DocLine" = T1."LineNum"
           AND O."StockEff" = 1) * IFNULL(NULLIF(T1."NumPerMsr",0), 1),

        NULL,

        T1."Quantity"

    FROM ORIN T0
    INNER JOIN RIN1 T1 ON T0."DocEntry" = T1."DocEntry"
    WHERE T0."DocDate" BETWEEN [%0] AND [%1]
      AND T0."CANCELED" = 'N'
      AND T0."CardCode" IN ('C0006588','C0006751')

) X

ORDER BY "التاريخ", "نوع المستند", "رقم المستند"
